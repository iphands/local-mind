> *basalt-line only* = for that commit object, three probes all come back negative — the ancestor's

> reading priority, never lower one. A reader who thinks "only three methods" and demotes item 1 is the

```bash

[ "$(wc -l < /tmp/opencode/paths-at-pin.txt)" -eq 862 ] || echo "NOTE: pin tree has $(wc -l < /tmp/opencode/paths-at-pin.txt) paths, expected 862"
n=0; missing=0; notin=0; tiermis=0; locmis=0
while IFS=$'\t' read -r p cloc ctier; do
  n=$((n+1))
  grep -qxF -- "$p" /tmp/opencode/paths-at-pin.txt || { echo "  MISSING_AT_PIN $p"; missing=$((missing+1)); continue; }
  m=$(awk -F'\t' -v p="$p" '$1==p{print $2"\t"$3; exit}' /tmp/opencode/man-tier-loc.txt)
  [ -z "$m" ] && { echo "  NOT_IN_MANIFEST $p"; notin=$((notin+1)); continue; }
  mt=${m%%$'\t'*}; ml=${m##*$'\t'}
  [ -n "$ctier" ] && [ "$ctier" != "$mt" ] && { echo "  TIER_MISMATCH $p claims=$ctier manifest=$mt"; tiermis=$((tiermis+1)); }
  [ "${cloc:-0}" -gt 0 ] 2>/dev/null && [ "$cloc" != "$ml" ] && { echo "  LOC_MISMATCH $p claims=$cloc manifest=$ml"; locmis=$((locmis+1)); }
done < readorder.tsv
echo "PATHS=$n MISSING=$missing NOT_IN_MANIFEST=$notin TIER_MISMATCH=$tiermis LOC_MISMATCH=$locmis"
[ "$n" -eq 10 ] || echo "INPUT WRONG: read $n rows, this read order has 10 — the result above is vacuous"
```
